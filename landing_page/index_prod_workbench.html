mascot_view.dart
Flutter / Mobile SDK 
import 'package:flutter/material.dart';
import 'package:rive/rive.dart';

class MascotWidget extends StatelessWidget {
  final int stateIndex; // 3 = loading, 4 = success
  const MascotWidget({required this.stateIndex});

  @override
  Widget build(BuildContext context) {
    return RiveAnimation.asset(
      'assets/mascot.riv',
      stateMachines: const ['MascotState'],
      onInit: (artboard) {
        final controller = StateMachineController.fromArtboard(
          artboard, 
          'MascotState',
        );
        artboard.addController(controller!);
        
        // Contrôle direct de l'état applicatif
        controller.findInput<double>('state')?.value = stateIndex.toDouble();
      },
    );
  }
}
